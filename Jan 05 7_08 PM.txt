import React, { useState } from 'react';

import SubscriptionForm from './SubscriptionForm';

import SubscriptionTable from './SubscriptionTable';

function NewspaperSubscriptionApp() {

  const [subscriptions, setSubscriptions] = useState([]);

  const [editIndex, setEditIndex] = useState(null);

  const addOrUpdateSubscription = (data) => {

    if (editIndex !== null) {

      const updated = [...subscriptions];

      updated[editIndex] = data;

      setSubscriptions(updated);

      setEditIndex(null);

    } else {

      setSubscriptions([...subscriptions, data]);

    }

  };

  const editSubscription = (index) => {

    setEditIndex(index);

  };

  const deleteSubscription = (index) => {

    const updated = subscriptions.filter((_, i) => i !== index);

    setSubscriptions(updated);

  };

  return (

    <div>

      <h2>Newspaper Subscription</h2>

      <SubscriptionForm

        onSubmit={addOrUpdateSubscription}

        editData={editIndex !== null ? subscriptions[editIndex] : null}

      />

      <SubscriptionTable

        subscriptions={subscriptions}

        onEdit={editSubscription}

        onDelete={deleteSubscription}

      />

    </div>

  );

}

export default NewspaperSubscriptionApp;

import React, { useEffect, useState } from 'react';

function SubscriptionForm({ onSubmit, editData }) {

  const [formData, setFormData] = useState({

    name: '',

    email: '',

    address: '',

    subscriptionType: 'Daily',

    paymentMethod: 'Credit Card'

  });

  useEffect(() => {

    if (editData) {

      setFormData(editData);

    }

  }, [editData]);

  const handleChange = (e) => {

    const { name, value } = e.target;

    setFormData({ ...formData, [name]: value });

  };

  const handleSubmit = (e) => {

    e.preventDefault();

    if (!formData.name || !formData.email || !formData.address) {

      alert('Please fill all required fields');

      return;

    }

    onSubmit(formData);

    setFormData({

      name: '',

      email: '',

      address: '',

      subscriptionType: 'Daily',

      paymentMethod: 'Credit Card'

    });

  };

  return (

    <form onSubmit={handleSubmit}>

      <h3>{editData ? 'Edit Subscription' : 'New Subscription'}</h3>

      <div>

        <label>Name:</label>

        <input name="name" value={formData.name} onChange={handleChange} />

      </div>

      <div>

        <label>Email:</label>

        <input name="email" value={formData.email} onChange={handleChange} />

      </div>

      <div>

        <label>Address:</label>

        <textarea name="address" value={formData.address} onChange={handleChange} />

      </div>

      <div>

        <label>Subscription Type:</label>

        <select name="subscriptionType" value={formData.subscriptionType} onChange={handleChange}>

          <option>Daily</option>

          <option>Weekly</option>

          <option>Monthly</option>

        </select>

      </div>

      <div>

        <label>Payment Method:</label>

        <select name="paymentMethod" value={formData.paymentMethod} onChange={handleChange}>

          <option>Credit Card</option>

          <option>Debit Card</option>

        </select>

      </div>

      <button type="submit">{editData ? 'Update' : 'Subscribe'}</button>

    </form>

  );

}

export default SubscriptionForm;

import React from 'react';

function SubscriptionTable({ subscriptions, onEdit, onDelete }) {

  if (subscriptions.length === 0) {

    return <p>No subscriptions available.</p>;

  }

  return (

    <>

      <h3>Subscription List</h3>

      <table border="1" cellPadding="5">

        <thead>

          <tr>

            <th>Name</th>

            <th>Email</th>

            <th>Address</th>

            <th>Plan</th>

            <th>Payment</th>

            <th>Actions</th>

          </tr>

        </thead>

        <tbody>

          {subscriptions.map((sub, index) => (

            <tr key={index}>

              <td>{sub.name}</td>

              <td>{sub.email}</td>

              <td>{sub.address}</td>

              <td>{sub.subscriptionType}</td>

              <td>{sub.paymentMethod}</td>

              <td>

                <button onClick={() => onEdit(index)}>Edit</button>

                <button onClick={() => onDelete(index)}>Delete</button>

              </td>

            </tr>

          ))}

        </tbody>

      </table>

    </>

  );

}

export default SubscriptionTable;
